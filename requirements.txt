# Streamlit Community Cloud / lightweight deployment dependency list.
#
# Notes
# -----
# * We pin the CPU wheel of PyTorch to avoid pulling the ~900 MB CUDA build
#   (which blows through the 1 GB Community Cloud image budget).
# * The trailing `.` installs this repository itself as the ``qqa`` package
#   so ``import qqa`` works inside the Streamlit app.
#
# Local development should still use ``uv sync --all-extras``; this file is
# for constrained deploy targets (Streamlit Community Cloud, Hugging Face
# Spaces, Render, etc.) and is intentionally valid for both pip and uv.

# Expose CPU Torch wheels as flat candidates while keeping PyPI as the sole
# package index.  This works with both pip and uv without weakening uv's
# dependency-confusion protection or selecting stale utility packages from a
# secondary index.
--find-links https://download.pytorch.org/whl/cpu/torch/
torch==2.13.0+cpu

numpy==2.5.2
networkx==3.6.1
scipy==1.18.1
matplotlib==3.11.1
tqdm==4.70.0

# Interactive plots + GUI (REQUIRED by app/_common.py at import time).
plotly==7.0.0
pandas==3.0.5
streamlit==1.63.0

# Install this repository so ``import qqa`` works inside the app.
# IMPORTANT: must be the LAST line. We bump pyproject.toml's ``version``
# field on every release that adds a public symbol (e.g. PA was added in
# 0.5.1, equilibrium-sample/genealogy/Muller-plot in 0.5.2). Without a
# version bump pip's resolver short-circuits when the previously
# installed ``qqa`` already matches the pinned spec — Streamlit
# Community Cloud has been observed to keep ``qqa==0.5.0`` from a stale
# build and silently skip reinstalling, which then makes
# ``qqa.population_annealing`` missing at runtime.
